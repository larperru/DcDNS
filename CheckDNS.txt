Once you've installed DcDNS, open Discord and press the keyboard shortcut: Shift+Ctrl+I.

This will open DevTools. Go to the Console tab.

Paste the command below. If you can't paste, type `allow pasting` first.

You should get the result: `Is Mullvad DNS: true`

If you get `false`, try reinstalling DcDNS. If that doesn't work, report it on GitHub under the Issues tab.

Then paste this command:

```javascript
(async function() {
    console.clear();
    try {
        const response = await fetch('https://edns.ip-api.com/json');
        const data = await response.json();

        const dnsIp = data.dns.ip || 'Unknown';
        const dnsGeo = data.dns.geo || 'Unknown';

        const mullvadProviders = [
            'mullvad',
            'm247',
            'datacamp',
            'datapacket',
            'tzulo',
            'creanova',
            'xtom',
            '31148',
            '208585',
            '207656'
        ];

        const geoLower = dnsGeo.toLowerCase();
        const isMullvad = mullvadProviders.some(provider => geoLower.includes(provider));

        console.table({
            "DNS Server": dnsIp,
            "DNS Provider": dnsGeo,
            "Is Mullvad DNS": isMullvad
        });
    } catch (error) {
        console.error('Verification failed:', error);
    }
})();
